# Задание 1

> Даны натуральные числа n и m. Напишите программу поиска всех пар дружественных чисел, лежащих в диапазоне от n до m. Два числа называются дружественными, если каждое из них равно сумме всех делителей другого (сами числа в указанные суммы не входят).

```pascal
program FriendlyNumbers;
var
	n, m, i, j, sum1, sum2: integer;

function SumOfDivisors(num: integer): integer;
var
	k, sum: integer
begin
	sum := 0;
	for k := 1 to num - 1 do
		if num mod k = 0 then
		sum := sum + k;
	SumOfDivisors := sum;
end;
	
begin
	write('Введите начало диапазона n: ');
	readln(n);
	write('Введите конец диапазона m: ');
	readln(m);
	
	writeln('Пары дружественных чисел в диапазоне [', n, ', ', m, ']:');
	  
	for i := n to m do
	begin
		sum1 := SumOfDivisors(i);
	    if (sum1 > i) and (sum1 <= m) then
	    begin
			sum2 := SumOfDivisors(sum1);
			if sum2 = i then
				writeln(i, ' и ', sum1, ' - дружественные числа');
		end;
	end;
end.
```

# Задание 2

> Напишите программу для графического изображения делимости чисел от 1 до n (n — исходное данное). В каждой строке надо печатать число и столько плюсов, сколько де-лителей у этого числа. Например, если исходное данное — число 4, то на экране должно быть напечатано:
> 
> 1+
> 
> 2++
> 
> 3++
> 
> 4+++


```pascal
program DivisorGraph;
var
	n, i, j, count: integer;

function CountDivisors(num: integer): integer;
var
	k, count: integer;
begin
	count := 0;
	for k := 1 to num do
    if num mod k = 0 then
		count := count + 1;
	CountDivisors := count;
end;

begin
	write('Введите число n: ');
	readln(n);
	
	writeln('Графическое изображение делимости чисел от 1 до ', n, ':');
	  
	for i := 1 to n do
	begin
		write(i, ' ');
		count := CountDivisors(i);
		for j := 1 to count do
			write('+');
		writeln;
	end;
end.
```

# Задание 3

> Даны натуральные числа a, b (a<b). Напишите программу нахождения всех простых чисел p, удовлетворяющих неравенству a ≤ p ≤ b.

```pascal
program PrimeNumbersInRange;
var
	a, b, i, j: integer;
	isPrime: boolean;

begin
	write('Введите начало диапазона a: ');
	readln(a);
	write('Введите конец диапазона b: ');
	readln(b);
	
	if a < 2 then a := 2;
	
	writeln('Простые числа в диапазоне [', a, ', ', b, ']:');
	
	for i := a to b do
	begin
		isPrime := true;
	    
		if i < 2 then
			isPrime := false
		else
			for j := 2 to trunc(sqrt(i)) do
				if i mod j = 0 then
				begin
					isPrime := false;
					break;
				end;
		
		if isPrime then
			write(i, ' ');
	end;
	writeln;
end.
```

# Задание 4

> Напишите программу поиска всех простых несократимых дробей, заключенных между 0 и 1, знаменатели которых не превышают 7 (дробь задается двумя натуральными числами — числителем и знаменателем).

```pascal
program IrreducibleFractions;
var
	numerator, denominator, i, gcd: integer;

function FindGCD(a, b: integer): integer;
begin
	while b <> 0 do
	begin
		gcd := a mod b;
		a := b;
		b := gcd;
	end;
	FindGCD := a;
end;

begin
	writeln('Простые несократимые дроби между 0 и 1 со знаменателем ≤ 7:');
	
	for denominator := 2 to 7 do
	begin
		for numerator := 1 to denominator - 1 do
	    begin
			gcd := FindGCD(numerator, denominator);
		      
			// Дробь несократима, если НОД = 1
			if gcd = 1 then
			writeln(numerator, '/', denominator);
		end;
	end;
end.
```
