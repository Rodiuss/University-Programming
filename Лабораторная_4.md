program Palindrom;

const endCycleNumber = 3;

var num: Integer;

function getDigitByIndex(number, index: integer): integer;
begin
  case index of
    0: getDigitByIndex := (number div 1000);
    1: getDigitByIndex := ((number div 100) mod 10);
    2: getDigitByIndex := ((number div 10) mod 10);
    3: getDigitByIndex := (number mod 10);
  end;
end;

begin
  num := readInteger('Введите натуральное четырёхзначное число:');
  
  for var i := 0 to endCycleNumber do
  begin
    if (getDigitByIndex(num, i)
      <> getDigitByIndex(num, endCycleNumber - i)) then
        begin
          writeln('Число не является палиндромом');
          exit;
        end;
  end;
  
  writeln('Число является палиндромом');
end.
