# Задание 1

> Даны натуральные числа n, k < 9999. Например, числа 6766, 5444, 0006, 0060 содержат ровно три озинаковых цифры.

```pascal
program DigitCount;

function getDgsInNum(number: Integer): array of Integer;
var divider: Integer;
begin
  divider := number > 999
    ? 1000
    : number > 99
      ? 100
      : number > 9
        ? 10
        : 1;
  case divider of
    1000: Result := [number div 1000, number mod 1000 div 100, number mod 100 div 10, number mod 10];
    100: Result := [number div 100, number mod 100 div 10, number mod 10];
    10: Result := [number div 10, number mod 10];
    1: Result := [number mod 10];
  end;    
end;

var number: Integer;

begin
  number := ReadInteger('Введите число: ');
  println(getDgsInNum(number));
end.
```
