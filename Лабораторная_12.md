# Задание 1

> Даны целые числа a₁, a₂, a₃. Получить целочисленную матрицу $\[b_{ij}\]i=1,2,3$ для которой $b_{ij} = a_i - 3a_j$.

```pascal
program MatrixFromValues;

var
  a: array [1..3] of Integer;
  b: array [1..3, 1..3] of Integer;
  i, j: Integer;

begin
  writeln('Введите три целых числа:');
  for i := 1 to 3 do
  begin
    write('a[', i, '] = ');
    readln(a[i]);
  end;
  
  for i := 1 to 3 do
    for j := 1 to 3 do
      b[i, j] := a[i] - 3 * a[j];
  
  writeln('Полученная матрица 3x3:');
  for i := 1 to 3 do
  begin
    for j := 1 to 3 do
      write(b[i, j]:5);
    writeln;
  end;
end.
```

# Задание 2

> Даны натуральное число $n$, матрица вещественных чисел размером $n x 9$. Найти среднее арифметическое каждого из столбцов, имеющих четные номера.

```pascal
program EvenColumnsAverage;

var
  n, i, j: Integer;
  matrix: array of array of Real;
  sum: Real;

begin
  write('Введите количество строк n: ');
  readln(n);
  
  SetLength(matrix, n);
  for i := 0 to n - 1 do
    SetLength(matrix[i], 9);
  
  writeln('Введите элементы матрицы ', n, 'x9:');
  for i := 0 to n - 1 do
    for j := 0 to 8 do
    begin
      write('matrix[', i, '][', j, '] = ');
      readln(matrix[i][j]);
    end;
  
  writeln('Средние арифметические столбцов с четными номерами:');
  for j := 1 to 8 do  // j от 1 до 8, так как нужны четные номера (2,4,6,8)
    if j mod 2 = 1 then  // индексация с 0, поэтому столбец 2 имеет индекс 1
    begin
      sum := 0;
      for i := 0 to n - 1 do
        sum := sum + matrix[i][j];
      writeln('Столбец ', j + 1, ': ', sum / n:0:2);
    end;
end.
```

# Задание 3

> Дана матрица вещественных чисел размера $n x m$, в которой не все элементы равны нулю. Получить новую матрицу путем деления всех элементов данной матрицы на её наибольший по модулю элемент.

```pascal
program NormalizeMatrix;

var
  n, m, i, j: Integer;
  matrix, newMatrix: array of array of Real;
  maxAbs, element: Real;

begin
  write('Введите количество строк n: ');
  readln(n);
  write('Введите количество столбцов m: ');
  readln(m);
  
  SetLength(matrix, n);
  SetLength(newMatrix, n);
  for i := 0 to n - 1 do
  begin
    SetLength(matrix[i], m);
    SetLength(newMatrix[i], m);
  end;
  
  writeln('Введите элементы матрицы ', n, 'x', m, ':');
  maxAbs := 0;
  for i := 0 to n - 1 do
    for j := 0 to m - 1 do
    begin
      write('matrix[', i, '][', j, '] = ');
      readln(matrix[i][j]);
      if abs(matrix[i][j]) > maxAbs then
        maxAbs := abs(matrix[i][j]);
    end;
  
  if maxAbs = 0 then
    writeln('Ошибка: все элементы матрицы равны нулю!')
  else
  begin
    writeln('Наибольший по модулю элемент: ', maxAbs:0:2);
    writeln('Новая матрица после деления:');
    
    for i := 0 to n - 1 do
      for j := 0 to m - 1 do
        newMatrix[i][j] := matrix[i][j] / maxAbs;
    
    for i := 0 to n - 1 do
    begin
      for j := 0 to m - 1 do
        write(newMatrix[i][j]:8:3);
      writeln;
    end;
  end;
end.
```

# Задание 4

> Дана матрица вещественных чисел размером $n x n$. В строках с отрицательным элементом на главной диагонали найти наибольший из всех элементов.

```pascal
program MaxInRowsWithNegativeDiagonal;

var
  n, i, j: Integer;
  matrix: array of array of Real;
  maxInRow: Real;
  found: Boolean;

begin
  write('Введите размер матрицы n: ');
  readln(n);
  
  SetLength(matrix, n);
  for i := 0 to n - 1 do
    SetLength(matrix[i], n);
  
  writeln('Введите элементы матрицы ', n, 'x', n, ':');
  for i := 0 to n - 1 do
    for j := 0 to n - 1 do
    begin
      write('matrix[', i, '][', j, '] = ');
      readln(matrix[i][j]);
    end;
  
  writeln('Результаты для строк с отрицательным элементом на главной диагонали:');
  found := False;
  
  for i := 0 to n - 1 do
  begin
    if matrix[i][i] < 0 then
    begin
      found := True;
      maxInRow := matrix[i][0];
      
      for j := 1 to n - 1 do
        if matrix[i][j] > maxInRow then
          maxInRow := matrix[i][j];
      
      writeln('Строка ', i + 1, 
              ' (диагональный элемент = ', matrix[i][i]:0:2, 
              '): максимальный элемент = ', maxInRow:0:2);
    end;
  end;
  
  if not found then
    writeln('Нет строк с отрицательным элементом на главной диагонали.');
end.
```

# Задание 5

> Таблица футбольного чемпионата задана квадратной матрицей порядка $n$, в которой все элементы, принадлежащие главной диагонали, равны нулю, а каждый элемент, не принадлежащий главной диагонали, равен 2, 1 или 0 (числу очков, набранных в игре: 2 -- выигрыш, 1 -- ничья, 0 -- проигрыш).

```pascal
program FootballChampionship;

var
  n, i, j: Integer;
  matrix: array of array of Integer;
  wins, losses, draws: Integer;
  teamsWithoutLosses: array of Integer;
  countWithoutLosses: Integer;
  hasTeamWithMoreThanHalfWins: Boolean;

begin
  write('Введите количество команд n: ');
  readln(n);
  
  SetLength(matrix, n);
  for i := 0 to n - 1 do
    SetLength(matrix[i], n);
  
  writeln('Введите результаты матчей (матрицу ', n, 'x', n, '):');
  writeln('0 - поражение, 1 - ничья, 2 - победа');
  writeln('Главная диагональ должна быть 0 (команда не играет сама с собой)');
  
  for i := 0 to n - 1 do
    for j := 0 to n - 1 do
    begin
      if i = j then
        matrix[i][j] := 0
      else
      begin
        repeat
          write('matrix[', i + 1, '][', j + 1, '] = ');
          readln(matrix[i][j]);
          if not (matrix[i][j] in [0, 1, 2]) then
            writeln('Ошибка! Введите 0, 1 или 2');
        until matrix[i][j] in [0, 1, 2];
      end;
    end;
  
  // a) Число команд с большим количеством побед, чем поражений
  writeln;
  writeln('a) Команды с большим количеством побед, чем поражений:');
  for i := 0 to n - 1 do
  begin
    wins := 0;
    losses := 0;
    
    for j := 0 to n - 1 do
    begin
      if i <> j then
      begin
        if matrix[i][j] = 2 then inc(wins)
        else if matrix[i][j] = 0 then inc(losses);
      end;
    end;
    
    if wins > losses then
      writeln('Команда ', i + 1, ': побед ', wins, ', поражений ', losses);
  end;
  
  // b) Номера команд, прошедших чемпионат без поражений
  writeln;
  writeln('б) Команды без поражений:');
  SetLength(teamsWithoutLosses, n);
  countWithoutLosses := 0;
  
  for i := 0 to n - 1 do
  begin
    losses := 0;
    for j := 0 to n - 1 do
      if (i <> j) and (matrix[i][j] = 0) then
        inc(losses);
    
    if losses = 0 then
    begin
      teamsWithoutLosses[countWithoutLosses] := i + 1;
      inc(countWithoutLosses);
    end;
  end;
  
  if countWithoutLosses = 0 then
    writeln('Нет команд без поражений')
  else
  begin
    write('Команды: ');
    for i := 0 to countWithoutLosses - 1 do
      write(teamsWithoutLosses[i], ' ');
    writeln;
  end;
  
  // c) Наличие команды, выигравшей более половины игр
  writeln;
  writeln('в) Команды, выигравшие более половины игр:');
  hasTeamWithMoreThanHalfWins := False;
  
  for i := 0 to n - 1 do
  begin
    wins := 0;
    for j := 0 to n - 1 do
      if (i <> j) and (matrix[i][j] = 2) then
        inc(wins);
    
    if wins > (n - 1) / 2 then
    begin
      writeln('Команда ', i + 1, ': ', wins, ' побед из ', n - 1, ' игр');
      hasTeamWithMoreThanHalfWins := True;
    end;
  end;
  
  if not hasTeamWithMoreThanHalfWins then
    writeln('Нет команд, выигравших более половины игр');
end.
```
