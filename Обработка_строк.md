# Задача 1

> Даны два предложения. В каком из них доля (в %) буквы б больше. (Определить функцию для расчета доли некоторой буквы в предложении.

```pascal
function LetterPercentage(s: string; ch: char): real;
var
  i, count: integer;
begin
  count := 0;
  for i := 1 to Length(s) do
    if s[i] = ch then
      Inc(count);
  Result := (count / Length(s)) * 100;
end;

var
  s1, s2: string;
  percent1, percent2: real;
  letter: char;
begin
  letter := 'б';
  
  Write('Введите первое предложение: ');
  ReadLn(s1);
  Write('Введите второе предложение: ');
  ReadLn(s2);
  
  percent1 := LetterPercentage(s1, letter);
  percent2 := LetterPercentage(s2, letter);
  
  WriteLn('Доля буквы "', letter, '" в первом предложении: ', percent1:0:2, '%');
  WriteLn('Доля буквы "', letter, '" во втором предложении: ', percent2:0:2, '%');
  
  if percent1 > percent2 then
    WriteLn('Больше в первом предложении')
  else if percent2 > percent1 then
    WriteLn('Больше во втором предложении')
  else
    WriteLn('Доли равны');
end.
```

# Задача 2

> Дан двумерный массив размером 5 x 5, заполненный буквами. Напечатать слово, образованное элементами массива, отмеченными звездочкой (рис. 12.9), при прочтении их:

```pascal
const
  N = 5;
type
  TMatrix = array[1..N, 1..N] of char;
  
procedure InitMatrix(var m: TMatrix);
begin
  m[1,1] := 'п'; m[1,2] := 'р'; m[1,3] := 'и'; m[1,4] := 'в'; m[1,5] := 'е';
  m[2,1] := 'т'; m[2,2] := 'с'; m[2,3] := 'т'; m[2,4] := 'р'; m[2,5] := 'о';
  m[3,1] := 'к'; m[3,2] := 'а'; m[3,3] := 'л'; m[3,4] := 'ь'; m[3,5] := 'к';
  m[4,1] := 'у'; m[4,2] := 'л'; m[4,3] := 'ь'; m[4,4] := 'т'; m[4,5] := 'а';
  m[5,1] := '!'; m[5,2] := '!'; m[5,3] := '!'; m[5,4] := '!'; m[5,5] := '!';
end;

function ReadHorizontal(const m: TMatrix): string;
var
  i, j: integer;
  s: string;
begin
  s := '';
  s := s + m[1,1] + m[1,3] + m[2,2] + m[3,3] + m[4,4] + m[5,5];
  Result := s;
end;

function ReadVertical(const m: TMatrix): string;
var
  i, j: integer;
  s: string;
begin
  s := '';
  for j := 1 to N do
    s := s + m[j, j];
  Result := s;
end;

var
  matrix: TMatrix;
begin
  InitMatrix(matrix);
  
  WriteLn('а) Слово (горизонтальное чтение): ', ReadHorizontal(matrix));
  WriteLn('б) Слово (вертикальное чтение): ', ReadVertical(matrix));
end.
```

# Задача 3

> Дан двумерный массив, элементами которого являются отдельные символы. Проверить, одинаковые ли строки массива с номерами n1 и n2?
> -  слева направо в каждой строке, начиная с первой;
> -  сверху вниз в каждом столбце, начиная с первого.

```pascal
const
  ROWS = 5;
  COLS = 5;
type
  TCharMatrix = array[1..ROWS, 1..COLS] of char;

function AreRowsEqual(const m: TCharMatrix; n1, n2: integer): boolean;
var
  j: integer;
begin
  Result := true;
  for j := 1 to COLS do
    if m[n1, j] <> m[n2, j] then
    begin
      Result := false;
      break;
    end;
end;

var
  matrix: TCharMatrix;
  n1, n2: integer;
  i, j: integer;
begin
  for i := 1 to ROWS do
    for j := 1 to COLS do
      matrix[i, j] := Chr(Ord('a') + Random(26));
  
  WriteLn('Матрица:');
  for i := 1 to ROWS do
  begin
    for j := 1 to COLS do
      Write(matrix[i, j], ' ');
    WriteLn;
  end;
  
  Write('Введите номера строк для сравнения (n1 n2): ');
  ReadLn(n1, n2);
  
  if (n1 < 1) or (n1 > ROWS) or (n2 < 1) or (n2 > ROWS) then
    WriteLn('Некорректные номера строк!')
  else if AreRowsEqual(matrix, n1, n2) then
    WriteLn('Строки ', n1, ' и ', n2, ' идентичны')
  else
    WriteLn('Строки ', n1, ' и ', n2, ' различны');
end.
```

# Задача 4

> Дан двумерный массив, элементами которого являются отдельные символы. Найти номер первого по порядку столбца массива, содержащего наибольшее число пробелов.

```pascal
const
  ROWS = 5;
  COLS = 5;
type
  TCharMatrix = array[1..ROWS, 1..COLS] of char;

function FindColumnWithMostSpaces(const m: TCharMatrix): integer;
var
  i, j, maxSpaces, currentSpaces, maxCol: integer;
begin
  maxSpaces := -1;
  maxCol := 0;
  
  for j := 1 to COLS do
  begin
    currentSpaces := 0;
    for i := 1 to ROWS do
      if m[i, j] = ' ' then
        Inc(currentSpaces);
    
    if currentSpaces > maxSpaces then
    begin
      maxSpaces := currentSpaces;
      maxCol := j;
    end;
  end;
  
  Result := maxCol;
end;

var
  matrix: TCharMatrix;
  i, j, resultCol: integer;
begin
  Randomize;
  for i := 1 to ROWS do
    for j := 1 to COLS do
      if Random(5) = 0 then
        matrix[i, j] := ' '
      else
        matrix[i, j] := Chr(Ord('a') + Random(26));
  
  // Вывод матрицы
  WriteLn('Матрица:');
  for i := 1 to ROWS do
  begin
    for j := 1 to COLS do
      if matrix[i, j] = ' ' then
        Write('_ ')
      else
        Write(matrix[i, j], ' ');
    WriteLn;
  end;
  
  resultCol := FindColumnWithMostSpaces(matrix);
  
  if resultCol > 0 then
    WriteLn('Первый столбец с наибольшим числом пробелов: ', resultCol)
  else
    WriteLn('Пробелов не найдено');
end.
```
