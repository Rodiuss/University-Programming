## Задание 1
> Напишите программу, которая находит максимальный и минимальный из элементов массива с чётными положительными значениями. Если в массиве нет элементов с чётными положительными значениями, нужно вывести сообщение об этом. Элементы массива вводятся с клавиатуры.

```pascal
program FindEvenPositiveMinMax;

var
  arr: array of Integer;
  n, i: Integer;
  minVal, maxVal: Integer;
  hasEvenPositive: Boolean;

begin
  n := ReadInteger('Введите размер массива:');
  arr := ReadArrInteger('Введите элементы массива:', n);
  
  hasEvenPositive := False;
  minVal := MaxInt;
  maxVal := -MaxInt;
  
  for i := 0 to n - 1 do
  begin
    if (arr[i] > 0) and (arr[i] mod 2 = 0) then
    begin
      hasEvenPositive := True;
      if arr[i] < minVal then minVal := arr[i];
      if arr[i] > maxVal then maxVal := arr[i];
    end;
  end;
  
  if hasEvenPositive then
    writeln('Минимальный четный положительный: ', minVal, 
            ', Максимальный четный положительный: ', maxVal)
  else
    writeln('В массиве нет четных положительных элементов');
end.
```

## Задание 2
> Введите массив с клавиатуры и найдите (за один проход) количество элементов, имеющих максимальное значение.

```pascal
program CountMaxElements;

var
  arr: array of Integer;
  n, i, maxVal, count: Integer;

begin
  n := ReadInteger('Введите размер массива:');
  arr := ReadArrInteger('Введите элементы массива:', n);
  
  maxVal := arr[0];
  count := 1;
  
  for i := 1 to n - 1 do
  begin
    if arr[i] > maxVal then
    begin
      maxVal := arr[i];
      count := 1;
    end
    else if arr[i] = maxVal then
      count += 1;
  end;
  
  writeln('Максимальное значение: ', maxVal);
  writeln('Количество элементов с максимальным значением: ', count);
end.
```

## Задание 3
> Заполните массив случайными числами в диапазоне 0..4 и выведите на экран номера всех элементов, значение которых равно Х (Х вводится с клавиатуры).

```pascal
program FindValuePositions;

var
  arr: array of Integer;
  n, i, x: Integer;
  found: Boolean;

begin
  n := ReadInteger('Введите размер массива:');
  SetLength(arr, n);
  
  Randomize;
  for i := 0 to n - 1 do
    arr[i] := Random(5); // 0..4
  
  writeln('Сгенерированный массив: ', arr);
  
  x := ReadInteger('Введите значение X для поиска (0..4):');
  
  found := False;
  write('Элемент ', x, ' найден на позициях: ');
  
  for i := 0 to n - 1 do
  begin
    if arr[i] = x then
    begin
      write(i, ' ');
      found := True;
    end;
  end;
  
  if not found then
    writeln('элемент не найден')
  else
    writeln;
end.
```

## Задание 4
> Заполните массив с чётным количеством элементов случайными числами и выполните реверс отдельно для первой и второй половин массива.

```pascal
program ReverseHalves;

var
  arr: array of Integer;
  n, i, temp: Integer;

begin
  repeat
    n := ReadInteger('Введите четный размер массива:');
  until n mod 2 = 0;
  
  SetLength(arr, n);
  
  Randomize;
  for i := 0 to n - 1 do
    arr[i] := Random(100); // 0..99
  
  writeln('Исходный массив: ', arr);
  
  for i := 0 to (n div 2) div 2 - 1 do
  begin
    temp := arr[i];
    arr[i] := arr[n div 2 - 1 - i];
    arr[n div 2 - 1 - i] := temp;
  end;
  
  for i := 0 to (n div 2) div 2 - 1 do
  begin
    temp := arr[n div 2 + i];
    arr[n div 2 + i] := arr[n - 1 - i];
    arr[n - 1 - i] := temp;
  end;
  
  writeln('Массив после реверса половин: ', arr);
end.
```

## Задание 5
> Найдите в массиве все простые числа и скопируйте их в новый массив (Элементы массива введите с клавиатуры)

```pascal
program FindPrimeNumbers;

function IsPrime(num: Integer): Boolean;
var
  i: Integer;
begin
  if num < 2 then
    Result := False
  else if num = 2 then
    Result := True
  else if num mod 2 = 0 then
    Result := False
  else
  begin
    Result := True;
    i := 3;
    while i * i <= num do
    begin
      if num mod i = 0 then
      begin
        Result := False;
        break;
      end;
      i += 2;
    end;
  end;
end;

var
  sourceArr, primeArr: array of Integer;
  n, i, primeCount: Integer;

begin
  n := ReadInteger('Введите размер массива:');
  SetLength(sourceArr, n);
  
  writeln('Введите элементы массива:');
  for i := 0 to n - 1 do
    sourceArr[i] := ReadInteger('arr[' + i + '] = ');
  
  primeCount := 0;
  for i := 0 to n - 1 do
    if IsPrime(sourceArr[i]) then
      primeCount += 1;
  
  SetLength(primeArr, primeCount);
  
  primeCount := 0;
  for i := 0 to n - 1 do
    if IsPrime(sourceArr[i]) then
    begin
      primeArr[primeCount] := sourceArr[i];
      primeCount += 1;
    end;
  
  writeln('Исходный массив: ', sourceArr);
  if primeCount > 0 then
    writeln('Простые числа: ', primeArr)
  else
    writeln('В массиве нет простых чисел');
end.
```
