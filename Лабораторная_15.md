# Задание 1

> Известны оценки каждого из 10 учеников класса по пяти предметам (Математика, Русский язык, Физика, Биология, Информатика). Определить среднюю оценку каждого ученика и всего класса. Вывести фамилии учеников, у которых средняя оценка выше средней по классу.

```pascal
program Lab15_Task1;

type
  Student = record
    LastName: string;
    Math, Russian, Physics, Biology, Informatics: integer;
    Average: real;
  end;

var
  Students: array[1..10] of Student;
  i: integer;
  ClassAverage, SumClass: real;

begin
  writeln('Введите данные о 10 учениках:');
  for i := 1 to 10 do
  begin
    writeln('Ученик №', i);
    write('Фамилия: ');
    readln(Students[i].LastName);
    write('Оценка по математике: ');
    readln(Students[i].Math);
    write('Оценка по русскому языку: ');
    readln(Students[i].Russian);
    write('Оценка по физике: ');
    readln(Students[i].Physics);
    write('Оценка по биологии: ');
    readln(Students[i].Biology);
    write('Оценка по информатике: ');
    readln(Students[i].Informatics);
    
    Students[i].Average := (Students[i].Math + Students[i].Russian + 
                           Students[i].Physics + Students[i].Biology + 
                           Students[i].Informatics) / 5;
    writeln;
  end;
  
  SumClass := 0;
  for i := 1 to 10 do
    SumClass := SumClass + Students[i].Average;
  ClassAverage := SumClass / 10;
  
  writeln('Результаты:');
  writeln('Средняя оценка по классу: ', ClassAverage:0:2);
  writeln;
  
  writeln('Средние оценки учеников:');
  for i := 1 to 10 do
    writeln(Students[i].LastName, ': ', Students[i].Average:0:2);
  writeln;
  
  writeln('Ученики со средней оценкой выше средней по классу:');
  for i := 1 to 10 do
    if Students[i].Average > ClassAverage then
      writeln(Students[i].LastName, ' (', Students[i].Average:0:2, ')');
end.
```

# Задание 2

> Известно расписание поездов, проходящих через станцию Вологда 1 (см. файл Расписание поездов_Вологда): номер поезда, назначение (откуда куда, например, Москва — Воркута), часы и минуты прибытия, часы и минуты отправления. Значения часов и минут целые, положительные; число часов не превышает 23, число минут — 59. Общее число проходящих поездов равно 10. Поезда приходят каждый день. По данному времени определить, какие поезда (номер и назначение) стоят в этот момент на станции.

```pascal
program schdl;

type
Time = record
  hour: Integer;
  min: Integer;
  constructor(arrayOfString: Array of String);
  begin
    self.hour := arrayOfString[0].toInteger;
    self.min := arrayOfString[1].toInteger;
  end;
end;

type
Schedule = record
  name: String;
  arrTime: Time;
  depTime: Time;
  
  constructor(arrayOfString: array of String);
  begin
    self.name := arrayOfString[0];
    self.arrTime := Time.Create(arrayOfString[1].Split(':'));
    self.depTime := Time.Create(arrayOfString[2].Split(':'));
  end;
  function isTime(yourTime: Time): Boolean;
  begin
    result := yourTime.hour.inRange(self.arrTime.hour, self.depTime.hour) and yourTime.min.inRange(self.arrTime.min, self.depTime.min);
  end;
  function toString: String; Override;
  begin
    result := String.Format('{0}: {1}:{2} - {3}:{4}', self.name, self.arrTime.hour,  self.arrTime.min,  self.depTime.hour,  self.depTime.min)
  end;
end;

var curText: Text;
var arrOfSchdl: Array of Schedule;
var searchTime: Time;

begin
  Assign(curText, 'file.txt');
  Reset(curText, Encoding.UTF8);
  arrOfSchdl := new Schedule[0];
  while not curText.Eof do
  begin
    arrOfSchdl := arrOfSchdl.Append(Schedule.Create(curText.readlnString.Split(','))).toArray;
  end;
  Close(curText);
  
  searchTime := Time.Create(ReadlnString('Введите чремя (часы минуты):').Split());
  writeln;
  foreach var i in arrOfSchdl.Where(x -> x.isTime(searchTime)) do
  begin
    println(i.ToString);
  end;
end.
```
