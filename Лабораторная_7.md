# Задание 1

> Вычислить и вывести с 7 знаками после запятой значение х, найденное по формуле:
> 
> $x = \frac{1 + \sin 4}{4 + \sin 1} + \frac{7 + \sin 5}{5 + \sin 7} + \frac{3 + \sin 2}{2 + \sin 3}$

```pascal
program CalculateFormula;
var
  x: real;

function CalculateX: real;
var
  numerator, denominator: real;
begin
  numerator := sqrt(3 + sqrt(6 + 4 * sqrt(2)));
  denominator := sqrt(3 - sqrt(6 + 4 * sqrt(2)));
  CalculateX := numerator / denominator;
end;

begin
  x := CalculateX;
  writeln('x = ', x:0:7);
end.
```

# Задание 2

> Реализовать и протестировать программу, выполняющую следующие операции: ввод с клавиатуры трех вещественных чисел, обозначающих длины отрезков, проверку того, что из них можно составить треугольник, вывод площади полученного треугольника или нуля, если треугольник составить нельзя. Для проверки возможности существования треугольника и для вычисления площади треугольника использовать функции.

```pascal
program TriangleCheck;
var
  a, b, c, area: real;

function CanFormTriangle(x, y, z: real): boolean;
begin
  CanFormTriangle := (x + y > z) and (x + z > y) and (y + z > x) and 
                     (x > 0) and (y > 0) and (z > 0);
end;

function CalculateTriangleArea(x, y, z: real): real;
var
  p: real;
begin
  p := (x + y + z) / 2;
  CalculateTriangleArea := sqrt(p * (p - x) * (p - y) * (p - z));
end;

begin
  write('Введите три длины отрезков: ');
  readln(a, b, c);
  
  if CanFormTriangle(a, b, c) then
  begin
    area := CalculateTriangleArea(a, b, c);
    writeln('Площадь треугольника: ', area:0:4);
  end
  else
    writeln('0');
end.
```

# Задание 3

> Известны следующие признаки делимости чисел:
> 
> a. на 2 необходимо, чтобы последняя цифра числа делилась на 2;
> 
> b. на 3 требуется, чтобы сумма цифр числа делилась на 3;
> 
> c. на 4 необходимо, чтобы число из последних двух цифр делилось на 4;
> 
> d. на 5 необходимо, чтобы последняя цифра числа была равна 0 или 5;
> 
> e. на 8 необходимо, чтобы число из 4 последних цифр делилось на 8;
> 
> f. на 9 необходимо, чтобы сумма цифр числа делилась на 9;
> 
> g. на 11 необходимо, чтобы разность между суммой цифр, стоящих на четных местах, и суммой цифр, стоящих на нечетных местах, делилась на 11.
> 
> Напишите процедуры проверки признаков делимости для числа n. Проверьте их для различных значений n.

```pascal
program DivisibilityTests;
var
  n: integer;

function IsDivisibleBy2(num: integer): boolean;
begin
  IsDivisibleBy2 := (num mod 10) mod 2 = 0;
end;

function IsDivisibleBy3(num: integer): boolean;
var
  sum, temp: integer;
begin
  sum := 0;
  temp := abs(num);
  while temp > 0 do
  begin
    sum := sum + (temp mod 10);
    temp := temp div 10;
  end;
  IsDivisibleBy3 := sum mod 3 = 0;
end;

function IsDivisibleBy4(num: integer): boolean;
begin
  IsDivisibleBy4 := (abs(num) mod 100) mod 4 = 0;
end;

function IsDivisibleBy5(num: integer): boolean;
begin
  IsDivisibleBy5 := (num mod 10 = 0) or (num mod 10 = 5);
end;

function IsDivisibleBy8(num: integer): boolean;
begin
  IsDivisibleBy8 := (abs(num) mod 10000) mod 8 = 0;
end;

function IsDivisibleBy9(num: integer): boolean;
var
  sum, temp: integer;
begin
  sum := 0;
  temp := abs(num);
  while temp > 0 do
  begin
    sum := sum + (temp mod 10);
    temp := temp div 10;
  end;
  IsDivisibleBy9 := sum mod 9 = 0;
end;

function IsDivisibleBy11(num: integer): boolean;
var
  sumEven, sumOdd, temp, position: integer;
begin
  sumEven := 0;
  sumOdd := 0;
  temp := abs(num);
  position := 1;
  
  while temp > 0 do
  begin
    if position mod 2 = 0 then
      sumEven := sumEven + (temp mod 10)
    else
      sumOdd := sumOdd + (temp mod 10);
    
    temp := temp div 10;
    position := position + 1;
  end;
  
  IsDivisibleBy11 := abs(sumEven - sumOdd) mod 11 = 0;
end;

procedure TestDivisibility(num: integer);
begin
  writeln('Число: ', num);
  writeln('Делится на 2: ', IsDivisibleBy2(num));
  writeln('Делится на 3: ', IsDivisibleBy3(num));
  writeln('Делится на 4: ', IsDivisibleBy4(num));
  writeln('Делится на 5: ', IsDivisibleBy5(num));
  writeln('Делится на 8: ', IsDivisibleBy8(num));
  writeln('Делится на 9: ', IsDivisibleBy9(num));
  writeln('Делится на 11: ', IsDivisibleBy11(num));
  writeln;
end;

begin
  TestDivisibility(123456);
  TestDivisibility(8910);
  TestDivisibility(121);
  TestDivisibility(72);
end.
```
