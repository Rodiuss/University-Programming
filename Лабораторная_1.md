# Задание 4

> Напишите программу, которая вычисляет и отображает размеры углов треугольника в градусах, исходя из введенных декартовых координат его вершин. 

```pascal
program TriangleAngles;

var
  x1, y1, x2, y2, x3, y3: real;  // Координаты вершин
  a, b, c: real;                    // Длины сторон
  alpha, beta, gamma: real;          // Углы в радианах
  alphaDeg, betaDeg, gammaDeg: real;  // Углы в градусах

// Функция вычисления расстояния между двумя точками
function Distance(x1, y1, x2, y2: real): real;
begin
  Distance := sqrt(sqr(x2 - x1) + sqr(y2 - y1));
end;

// Функция перевода углов из радиан в градусы
function toDeg(angular: real): real;
begin
  toDeg := angular * 180 / pi;
end;

begin
  // Ввод координат вершин треугольника
  writeln('Введите координаты первой вершины (x1 y1):');
  readln(x1, y1);
  writeln('Введите координаты второй вершины (x2 y2):');
  readln(x2, y2);
  writeln('Введите координаты третьей вершины (x3 y3):');
  readln(x3, y3);
  writeln;
  // Вычисление длин сторон треугольника

  a := Distance(x2, y2, x3, y3);  // сторона против угла alpha
  b := Distance(x1, y1, x3, y3);  // сторона против угла beta
  c := Distance(x1, y1, x2, y2);  // сторона против угла gamma
  
  // Проверка существования треугольника (неравенство треугольника)
  if (a + b <= c) or (a + c <= b) or (b + c <= a) then
  begin
    writeln('Ошибка: треугольник с такими координатами не существует!');
    exit;
  end;
  
  // Вычисление углов по теореме косинусов
  // cos(alpha) = (b? + c? - a?) / (2bc)
  alpha := arccos((b*b + c*c - a*a) / (2*b*c));
  // cos(beta) = (a? + c? - b?) / (2ac)
  beta := arccos((a*a + c*c - b*b) / (2*a*c));
  // gamma = pi - alpha - beta (сумма углов треугольника = pi радиан)
  gamma := pi - alpha - beta;
  
  // Перевод углов из радиан в градусы
  alphaDeg := toDeg(alpha);
  betaDeg := toDeg(beta);
  gammaDeg := toDeg(gamma);
  
  // Вывод результатов
  writeln('Размеры углов треугольника:');
  writeln('Угол при вершине (x1,y1): ', alphaDeg:0:2, ' градусов');
  writeln('Угол при вершине (x2,y2): ', betaDeg:0:2, ' градусов');
  writeln('Угол при вершине (x3,y3): ', gammaDeg:0:2, ' градусов');
  writeln;

  // Проверка: сумма углов должна быть 180 градусов
  writeln('Сумма углов: ', (alphaDeg + betaDeg + gammaDeg):0:2, ' градусов');
end.
```

# Задание 5

> 5.	Дано трехзначное число (ввод с клавиатуры). Определите:
>
> a.	число, полученное перестановкой цифр сотен и десятков;
>
> b.	четырехзначное число, полученное приписыванием цифры единиц в качестве цифры тысяч (например, из числа 137 необходимо получить 7137).
>
> Числа вывести столбиком.

```pascal
program TransformNumber;

var
  num, hundreds, tens, units, result1, result2: integer;
begin
  // Ввод трёхзначного числа
  writeln('Введите трёхзначное число:');
  readln(num);

  // Проверка, что число действительно трёхзначное
  if (num < 100) or (num > 999) then
  begin
    writeln('Ошибка: число должно быть трёхзначным!');
    exit;
  end;

  // Выделяем цифры числа
  hundreds := num div 100;        // цифра сотен
  tens := (num div 10) mod 10;   // цифра десятков
  units := num mod 10;            // цифра единиц

  // Получаем число перестановкой сотен и десятков
  result1 := tens * 100 + hundreds * 10 + units;
  
  // Получаем четырёхзначное число, приписывая цифру единиц в качестве тысячи
  result2 := units * 1000 + num;

  // Выводим результаты столбиком
  writeln(result1);
  writeln(result2);
end.
```

# Задание 6

> Сумма первых п членов арифметической прогрессии вычисляется по формуле: $S_{n}=(a_{1} + a_{n}) \cdot n / 2$. Даны $a_{1}$ и $d$. Найдите (экспериментальным путем) $n$, при котором значение $S_{n}$ выходит за диапазон типа `Integer`.


```pascal
program FindOverflow;

var
  a1, d: integer;
  n: int64 := 1;
  an, Sn: int64;
  overflow: boolean := false;

begin
  writeln('Введите первый член арифметической прогрессии (a1):');
  readln(a1);
  writeln('Введите разность арифметической прогрессии (d):');
  readln(d);
  
  while not overflow do
  begin
    an := a1 + (n - 1) * d;
    Sn := (a1 + an) * n div 2;

    // Проверяем, не вышло ли Sn за границы Integer
    if (Sn < integer.MinValue) or (Sn > integer.MaxValue) then
      overflow := true
    else
      n := n + 1;
  end;

  // Вывод результата
  writeln('Переполнение происходит при n = ', n);
  writeln('При этом:');
  writeln('  a1 = ', a1);
  writeln('  d  = ', d);
  writeln('  an = ', a1 + (n - 1) * d);
  writeln('  Sn = (a1 + an) * n / 2 = (', a1, ' + ', a1 + (n - 1) * d, ') * ', n, ' / 2');

  // Показываем, что Sn уже не помещается в Integer
  writeln('Значение Sn выходит за диапазон [', integer.MinValue, '..', integer.MaxValue, ']');
end.
```
