# Задание 1

> Дано предложение. Напечатать все его символы, предшествующие первой запятой. Рассмотреть два случая:
> 1) известно, что в предложении запятые имеются;
> 2) в предложении запятых может не быть.

```pascal
program BeforeFirstComma;

var
  s: string;
  i, commaPos: Integer;

begin
  write('Введите предложение: ');
  readln(s);
  
  commaPos := 0;
  for i := 1 to Length(s) do
    if s[i] = ',' then
    begin
      commaPos := i;
      break;
    end;
  
  if commaPos > 0 then
  begin
    writeln('Символы перед первой запятой:');
    write('"');
    for i := 1 to commaPos - 1 do
      write(s[i]);
    writeln('"');
  end
  else
    writeln('В предложении нет запятых');
end.
```

# Задание 2

> Даны два слова. Определить, сколько начальных букв первого слова совпадает с начальными буквами второго слова. Рассмотреть два случая:
> 1) известно, что слова разные;
> 2) слова могут быть одинаковыми.

```pascal
program MatchingPrefix;

var
  word1, word2: string;
  i, minLength, matchCount: Integer;

begin
  write('Введите первое слово: ');
  readln(word1);
  write('Введите второе слово: ');
  readln(word2);
  
  minLength := Length(word1);
  if Length(word2) < minLength then
    minLength := Length(word2);
  
  matchCount := 0;
  
  for i := 1 to minLength do
    if word1[i] = word2[i] then
      matchCount := matchCount + 1
    else
      break;
  
  writeln('Количество совпадающих начальных букв: ', matchCount);
  
  if word1 = word2 then
    writeln('Слова одинаковые')
  else if matchCount = minLength then
    writeln('Одно слово является началом другого');
end.
```

# Задание 3
> Дано предложение. Все его символы, стоящие на третьем, шестом, девятом и т. д. местах, заменить буквой а.

```pascal
program ReplaceEveryThird;

var
  s: string;
  i: Integer;

begin
  write('Введите предложение: ');
  readln(s);
  
  writeln('Исходное предложение: ', s);
  
  for i := 1 to Length(s) do
    if i mod 3 = 0 then
      s[i] := 'a';
  
  writeln('Измененное предложение: ', s);
end.
```

# Задание 4

> Дан текст. Определить количество цифр в нем.

```pascal
program CountDigits;

var
  s: string;
  i, digitCount: Integer;

begin
  write('Введите текст: ');

```

# Задание 5

> Дан текст, в начале которого имеются пробелы и в котором имеются цифры. Найти порядковый номер максимальной цифры, начиная счет с первого символа, не являющегося пробелом. Если максимальных цифр несколько, то должен быть найден номер первой из них.

```pascal
program FindMaxDigitPosition;

var
  s: string;
  i, startPos, maxDigit, maxDigitPos: Integer;
  digitFound: Boolean;

begin
  write('Введите текст: ');
  readln(s);
  
  startPos := 1;
  while (startPos <= Length(s)) and (s[startPos] = ' ') do
    startPos := startPos + 1;
  
  if startPos > Length(s) then
  begin
    writeln('Текст состоит только из пробелов');
    exit;
  end;
  
  maxDigit := -1;
  maxDigitPos := -1;
  digitFound := False;
  
  for i := startPos to Length(s) do
    if s[i] in ['0'..'9'] then
    begin
      digitFound := True;
      if Ord(s[i]) - Ord('0') > maxDigit then
      begin
        maxDigit := Ord(s[i]) - Ord('0');
        maxDigitPos := i;
      end;
    end;
  
  if digitFound then
  begin
    writeln('Максимальная цифра: ', maxDigit);
    writeln('Позиция максимальной цифры (от начала строки): ', maxDigitPos);
    writeln('Позиция от первого не-пробела: ', maxDigitPos - startPos + 1);
  end
  else
    writeln('В тексте нет цифр');
end.
```

# Задание 6

> Дана строка текста, в котором нет начальных и конечных пробелов. Необходимо изменить ее так, чтобы длина строки стала равна заданной длине (предполагается, что требуемая длина не меньше исходной). Это следует сделать путем вставки между словами дополнительных пробелов. Количество пробелов между отдельными словами должно отличаться не более чем на 1.

```pascal
program AdjustStringLength;

var
  s, result: string;
  targetLength, currentLength, wordCount, i: Integer;
  spacesNeeded, baseSpaces, extraSpaces: Integer;
  words: array of string;
  spacesBetween: array of Integer;

begin
  write('Введите строку без начальных и конечных пробелов: ');
  readln(s);
  write('Введите требуемую длину строки: ');
  readln(targetLength);
  
  currentLength := Length(s);
  
  if targetLength < currentLength then
  begin
    writeln('Ошибка: требуемая длина меньше исходной');
    exit;
  end;
  
  if targetLength = currentLength then
  begin
    writeln('Строка уже имеет требуемую длину');
    exit;
  end;
  
  words := s.ToWords;
  wordCount := words.Length;
  
  if wordCount < 2 then
  begin
    writeln('Строка содержит только одно слово. Нельзя вставить пробелы между словами.');
    exit;
  end;
  
  spacesNeeded := targetLength - currentLength;
  
  baseSpaces := spacesNeeded div (wordCount - 1);
  
  extraSpaces := spacesNeeded mod (wordCount - 1);
  
  SetLength(spacesBetween, wordCount - 1);
  
  for i := 0 to wordCount - 2 do
  begin
    spacesBetween[i] := baseSpaces;
    
    if i < extraSpaces then
      spacesBetween[i] := spacesBetween[i] + 1;
  end;

  result := words[0];
  
  for i := 0 to wordCount - 2 do
  begin
    result := result + ' ' + StringOfChar(' ', spacesBetween[i]) + words[i + 1];
  end;
  
  writeln('Результат:');
  writeln(result);
  writeln('Длина результата: ', Length(result));
end.
```

# Задание 7

> Дано натуральное число n (n <= 1000). Напечатать это число русскими словами (тринадцать, сто пять, двести сорок один, тысяча и т. д.)

```pascal
program NumberToWords;

function DigitToWords(digit: Integer): string;
begin
  case digit of
    1: Result := 'один';
    2: Result := 'два';
    3: Result := 'три';
    4: Result := 'четыре';
    5: Result := 'пять';
    6: Result := 'шесть';
    7: Result := 'семь';
    8: Result := 'восемь';
    9: Result := 'девять';
    0: Result := '';
  end;
end;

function TeenToWords(num: Integer): string;
begin
  case num of
    10: Result := 'десять';
    11: Result := 'одиннадцать';
    12: Result := 'двенадцать';
    13: Result := 'тринадцать';
    14: Result := 'четырнадцать';
    15: Result := 'пятнадцать';
    16: Result := 'шестнадцать';
    17: Result := 'семнадцать';
    18: Result := 'восемнадцать';
    19: Result := 'девятнадцать';
  end;
end;

function TensToWords(tens: Integer): string;
begin
  case tens of
    2: Result := 'двадцать';
    3: Result := 'тридцать';
    4: Result := 'сорок';
    5: Result := 'пятьдесят';
    6: Result := 'шестьдесят';
    7: Result := 'семьдесят';
    8: Result := 'восемьдесят';
    9: Result := 'девяносто';
  end;
end;

function HundredsToWords(hundreds: Integer): string;
begin
  case hundreds of
    1: Result := 'сто';
    2: Result := 'двести';
    3: Result := 'триста';
    4: Result := 'четыреста';
    5: Result := 'пятьсот';
    6: Result := 'шестьсот';
    7: Result := 'семьсот';
    8: Result := 'восемьсот';
    9: Result := 'девятьсот';
  end;
end;

var
  n: Integer;
  result: string;
  hundreds, tens, units: Integer;

begin
  write('Введите натуральное число (1-1000): ');
  readln(n);
  
  if (n < 1) or (n > 1000) then
  begin
    writeln('Число должно быть в диапазоне от 1 до 1000');
    exit;
  end;
  
  if n = 1000 then
  begin
    writeln('тысяча');
    exit;
  end;
  
  hundreds := n div 100;
  tens := (n mod 100) div 10;
  units := n mod 10;
  
  result := '';
  
  if hundreds > 0 then
  begin
    result := result + HundredsToWords(hundreds);
    if (tens > 0) or (units > 0) then
      result := result + ' ';
  end;
  
  if tens = 1 then
  begin
    result := result + TeenToWords(10 + units);
  end
  else
  begin
    if tens > 1 then
    begin
      result := result + TensToWords(tens);
      if units > 0 then
        result := result + ' ';
    end;
    
    if units > 0 then
    begin
      if (hundreds = 0) and (tens = 0) and (units in [1, 2]) then
      begin
        if units = 1 then
          result := result + 'один'
        else
          result := result + 'два';
      end
      else
      begin
        result := result + DigitToWords(units);
      end;
    end;
  end;
  
  if result = '' then
    result := 'ноль';
  
  writeln(result);
end.
```
