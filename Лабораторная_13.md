# Задание 1

> Напишите программу, которая выводит True, если в строке буква А встречается чаще, чем буква B, и False в противном случае.

```pascal
program CompareAB;

var
  s: string;
  countA, countB, i: Integer;

begin
  write('Введите строку: ');
  readln(s);
  
  countA := 0;
  countB := 0;
  
  for i := 1 to Length(s) do
  begin
    if (s[i] = 'A') or (s[i] = 'a') then
      countA := countA + 1
    else if (s[i] = 'B') or (s[i] = 'b') then
      countB := countB + 1;
  end;
  
  writeln('Количество A: ', countA);
  writeln('Количество B: ', countB);
  writeln('A встречается чаще, чем B: ', countA > countB);
end.
```

# Задание 2

> Подсчитайте количество гласных латинских букв в строке.

```pascal
program CountVowels;

var
  s: string;
  count, i: Integer;

begin
  write('Введите строку: ');
  readln(s);
  
  count := 0;
  
  for i := 1 to Length(s) do
  begin
    case s[i] of
      'A', 'a', 'E', 'e', 'I', 'i', 'O', 'o', 'U', 'u', 'Y', 'y':
        count := count + 1;
    end;
  end;
  
  writeln('Количество гласных латинских букв: ', count);
end.
```

# Задание 4

> Дан текст. Выведите слова, встречающиеся в тексте по одному разу.

```pascal
program UniqueWords;

var
  text: string;
  words, uniqueWords: array of string;
  i, j: Integer;
  isUnique: Boolean;

begin
  write('Введите текст: ');
  readln(text);
  
  words := text.ToLower.ToWords;
  
  SetLength(uniqueWords, 0);
  
  for i := 0 to words.Length - 1 do
  begin
    isUnique := True;
    
    for j := 0 to words.Length - 1 do
      if (i <> j) and (words[i] = words[j]) then
      begin
        isUnique := False;
        break;
      end;
    
    if isUnique then
    begin
      SetLength(uniqueWords, uniqueWords.Length + 1);
      uniqueWords[uniqueWords.Length - 1] := text.ToWords[i];
    end;
  end;
  
  writeln('Слова, встречающиеся по одному разу:');
  if uniqueWords.Length > 0 then
    for i := 0 to uniqueWords.Length - 1 do
      writeln(uniqueWords[i])
  else
    writeln('Таких слов нет');
end.
```

# Задание 5

> Дан текст. Написать программу проверки правильности написания сочетаний «жи», «ши», «ча», «ща», «чу» и «щу». Если в тексте есть ошибки, вывести сообщение об этом. Затем исправить ошибки и вывести правильно написанный текст.

```pascal
program CorrectSpelling;

var
  text, correctedText: string;
  i: Integer;
  hasErrors: Boolean;

begin
  write('Введите текст: ');
  readln(text);
  
  correctedText := text;
  hasErrors := False;
  
  writeln('Поиск ошибок в сочетаниях "жи/ши", "ча/ща", "чу/щу":');
  
  for i := 1 to Length(text) - 1 do
  begin
    if ((text[i] = 'ж') or (text[i] = 'Ж') or 
        (text[i] = 'ш') or (text[i] = 'Ш')) and 
       ((text[i+1] = 'ы') or (text[i+1] = 'Ы')) then
    begin
      writeln('Ошибка в позиции ', i, ': "', text[i], text[i+1], '"');
      hasErrors := True;
      correctedText[i+1] := if text[i+1] = 'ы' then 'и' else 'И';
    end;
    
    if ((text[i] = 'ч') or (text[i] = 'Ч') or 
        (text[i] = 'щ') or (text[i] = 'Щ')) and 
       ((text[i+1] = 'я') or (text[i+1] = 'Я')) then
    begin
      writeln('Ошибка в позиции ', i, ': "', text[i], text[i+1], '"');
      hasErrors := True;
      correctedText[i+1] := if text[i+1] = 'я' then 'а' else 'А';
    end;
    
    if ((text[i] = 'ч') or (text[i] = 'Ч') or 
        (text[i] = 'щ') or (text[i] = 'Щ')) and 
       ((text[i+1] = 'ю') or (text[i+1] = 'Ю')) then
    begin
      writeln('Ошибка в позиции ', i, ': "', text[i], text[i+1], '"');
      hasErrors := True;
      correctedText[i+1] := if text[i+1] = 'ю' then 'у' else 'У';
    end;
  end;
  
  if not hasErrors then
    writeln('Ошибок не найдено')
  else
  begin
    writeln('Исправленный текст:');
    writeln(correctedText);
  end;
end.
```

# Задание 6

> Проверьте, правильно ли в заданном тексте расставлены круглые скобки.

```pascal
program CheckParentheses;

var
  text: string;
  balance, i: Integer;
  isCorrect: Boolean;

begin
  write('Введите текст: ');
  readln(text);
  
  balance := 0;
  isCorrect := True;
  
  for i := 1 to Length(text) do
  begin
    if text[i] = '(' then
      balance := balance + 1
    else if text[i] = ')' then
      balance := balance - 1;
    
    if balance < 0 then
    begin
      isCorrect := False;
      break;
    end;
  end;
  
  if balance <> 0 then
    isCorrect := False;
  
  if isCorrect then
    writeln('Скобки расставлены правильно')
  else
    writeln('Скобки расставлены неправильно');
end.
```

# Задание 7

> Даны три строки. Определите, можно ли из символов первых двух строк получить третью строку.

```pascal
program CanFormString;

function CanForm(str1, str2, target: string): Boolean;
var
  charCount: array ['A'..'z'] of Integer;
  c: Char;
  i: Integer;
begin
  for c := 'A' to 'z' do
    charCount[c] := 0;
  
  for i := 1 to Length(str1) do
    charCount[str1[i]] := charCount[str1[i]] + 1;
  
  for i := 1 to Length(str2) do
    charCount[str2[i]] := charCount[str2[i]] + 1;
  
  for i := 1 to Length(target) do
  begin
    charCount[target[i]] := charCount[target[i]] - 1;
    if charCount[target[i]] < 0 then
    begin
      Result := False;
      exit;
    end;
  end;
  
  Result := True;
end;

var
  str1, str2, str3: string;

begin
  writeln('Введите три строки:');
  write('Первая строка: ');
  readln(str1);
  write('Вторая строка: ');
  readln(str2);
  write('Третья строка: ');
  readln(str3);
  
  if CanForm(str1, str2, str3) then
    writeln('Третью строку МОЖНО получить из символов первых двух')
  else
    writeln('Третью строку НЕЛЬЗЯ получить из символов первых двух');
end.
```
