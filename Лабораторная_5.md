# Задание 2

> Напишите программу определения того, сколько раз в натуральном числе встречается его максимальная цифра. Например, в числе 581088 — 3 раза, в числе 4537 — 1 раз.

```pascal
program FindMaxDigitCount;
var
	number, temp, maxDigit, count, currentDigit: integer;
begin
	number := readInteger('Введите натуральное число:');
	
	// Находим максимальную цифру
	temp := number;
	maxDigit := 0;
	
	while temp > 0 do
	begin
		currentDigit := temp mod 10;
		if currentDigit > maxDigit then
		  maxDigit := currentDigit;
		temp := temp div 10;
	end;
	
	// Считаем, сколько раз встречается максимальная цифра
	temp := number;
	count := 0;
	
	while temp > 0 do
	begin
		currentDigit := temp mod 10;
		if currentDigit = maxDigit then
			count := count + 1;
		temp := temp div 10;
	end;
	
	writeln('Максимальная цифра: ', maxDigit);
	writeln('Встречается раз: ', count);
end.
```

# Задание 3

> Последовательность Фибоначчи образуется так: первый и второй члены последовательности равны 1, каждый следующий равен сумме двух предыдущих (1, 1, 2, 3, 5, 8, 13, ...). Найти первое число в последовательности Фибоначчи, большее n (значение n вводится с клавиатуры; n > 1).

```pascal
program FindFirstFibonacciGreaterThanN;
var
	n, a, b, next: integer;
begin
	write('Введите n (n > 1): ');
	readln(n);
	
	a := 1;
	b := 1;
	next := b;
	
	while next <= n do
	begin
		next := a + b;
		a := b;
		b := next;
	end;
	
	writeln('Первое число Фибоначчи, большее ', n, ', равно ', next);
end.
```

# Задание 4

> Баллы, полученные на по пятиборью каждым из 8 спортсменов по каждому виду спорта, представленны в виде таблицы
> |      | Вид спорта |  |  |  |
> | :---: | :---: | :---: | :---: | :---: |
> | Спортсмен  | 1 | 2 | ... | 5 |
> | 1 | 876 | 655 |  | 604 |
> | 2 | 744 | 634 |  | 780 |
> | ... |  | ... |  | ... |
> | 8 | 897 | 880 |  | 798 |
> 
> Составить программу, которая запрашивает вид спорта, затем результаты спортсменов и отображает внесенные данные в таблице (в столбце с соответствующим номером или названием вида спорта).

```pascal
program PentathlonSimple;
const
  N = 8; // спортсменов
  M = 5; // видов спорта
var
  results: array[1..N, 1..M] of integer;
  i, j, sportChoice: integer;

begin
  for i := 1 to N do
    for j := 1 to M do
      results[i, j] := 0;
  
  writeln('=== ВВОД РЕЗУЛЬТАТОВ ПЯТИБОРЬЯ ===');
  writeln;
  
  repeat
    writeln('Виды спорта:');
    writeln('1 - Вид 1');
    writeln('2 - Вид 2'); 
    writeln('3 - Вид 3');
    writeln('4 - Вид 4');
    writeln('5 - Вид 5');
    write('Выберите вид спорта (0 - выход): ');
    readln(sportChoice);
    
    if (sportChoice >= 1) and (sportChoice <= M) then
    begin
		writeln;
		writeln('Ввод результатов для Вида ', sportChoice, ':');
		
		// Ввод результатов
		for i := 1 to N do
		begin
			write('Спортсмен ', i, ': ');
			readln(results[i, sportChoice]);
		end;
		
		// Вывод обновленной таблицы
		writeln;
		writeln('ТАБЛИЦА РЕЗУЛЬТАТОВ:');
		write('Спортсмен':10);
		for j := 1 to M do
			write('Вид ' + j:8);
		writeln;
		
		for i := 1 to N do
		begin
			write(i:8, ' |');
			for j := 1 to M do
			begin
				if results[i, j] = 0 then
					write('---':8)
				else
					write(results[i, j]:8);
			end;
			writeln;
		end;
		writeln;
	end;
	
	until sportChoice = 0;
	
	writeln('Программа завершена.');
end.
```
